// Initializing Nova High-Throughput HTTP Engine
import { NovaServer, RateLimiter, JWTAuth } from '@nova/core';

const app = new NovaServer({
  port: 8080,
  clustering: 'auto',
  telemetry: true
});

app.use(RateLimiter({ rpm: 5000 }));
app.post('/api/v1/predict', JWTAuth(), async (ctx) => {
  const result = await ctx.ai.synthesize({
    prompt: ctx.request.body.prompt,
    mode: 'deterministic'
  });
  return ctx.json({ status: 200, data: result });
});

await app.listen();
console.log('🚀 Nova microservice running on port 8080');
